# OpenFX reference host (from the OpenFX HostSupport examples) used to load and render the Lukuluku OFX plugin in tests.
cmake_minimum_required(VERSION 3.20)
project(ofxtesthost C CXX)
set(CMAKE_CXX_STANDARD 11)
set(HS ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/openfx/HostSupport)
set(OFXI ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/openfx/include)
file(GLOB HS_SRC ${HS}/src/*.cpp)
add_library(expat STATIC ${HS}/expat-2.0.1/lib/xmlparse.c ${HS}/expat-2.0.1/lib/xmlrole.c ${HS}/expat-2.0.1/lib/xmltok.c)
target_compile_definitions(expat PUBLIC HAVE_MEMMOVE XML_NS XML_DTD XML_CONTEXT_BYTES=1024 BYTEORDER=1234 XML_STATIC)
target_include_directories(expat PUBLIC ${HS}/expat-2.0.1/lib)
add_executable(ofxtesthost hostDemo.cpp hostDemoClipInstance.cpp hostDemoEffectInstance.cpp hostDemoHostDescriptor.cpp hostDemoParamInstance.cpp ${HS_SRC})
target_include_directories(ofxtesthost PRIVATE ${HS}/include ${OFXI} .)
target_compile_options(ofxtesthost PRIVATE -w -fpermissive)
target_link_libraries(ofxtesthost expat dl)
